DLアーキテクチャーを色々試してみる

import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.datasets import load_wine
from sklearn.preprocessing import StandardScaler


class MLP(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(13, 64)
        self.fc2 = nn.Linear(64, 3)

    def forward(self, x):
        x = self.fc1(x)
        x = torch.relu(x)
        x = self.fc2(x)
        return x


model = MLP()

data = load_wine()

scaler = StandardScaler()
x = scaler.fit_transform(data.data)
x = torch.tensor(x, dtype=torch.float32)

target = torch.tensor(data.target, dtype=torch.long)
criterion = nn.CrossEntropyLoss()

optimizer = optim.Adam(model.parameters(), lr=1e-4)
for _ in range(10):
    logits = model(x)

    loss = criterion(logits, target)

    optimizer.zero_grad()

    loss.backward()

    optimizer.step()

    print(loss.item())

これに適したDatasetはFashionMNISTらしいので，これを使う．

まずCNNの順伝播を復習する．Xは入力とし，畳み込み層は１層とする． \begin{align} Z_{kij} &= b_k + \sum_{c =1}^{C_{\text{in}}}\sum_{u = 0}^{K_h - 1} \sum_{v = 0}^{K_w - 1} W_{kcuv} X_{c,i +u,j+v}\\ A_{kij} &= \text{ReLU}(Z_{kij}) \\ P_{kij} &= \max_{0 \le u,v < 2} A_{k,2i+u,2j + v} \\ p &= \text{Flatten}(P_{kij})\\ z_1 &= W_1 p + b_1\\ h_1 &= \text{ReLU}(z_1) \\ z_2 &= W_2 h_1 + b_2 \\ \end{align} 損失関数は分類なのでCrossEntropyである． クラスは次のように定義される

import torch
import torch.nn as nn
import torch.optim as optim


class CNN(nn.Module):
    def __init__(self):
        super().__init__()

        self.conv1 = nn.Conv2d(in_channels=1, out_channels=16, kernel_size=3, padding=1)

        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)

        self.fc1 = nn.Linear(16 * 14 * 14, 64)
        self.fc2 = nn.Linear(64, 10)

    def forward(self, x):
        Z = self.conv1(x)
        A = torch.relu(Z)
        P = self.pool(A)
        p = torch.flatten(P, 1)
        z1 = self.fc1(p)
        h1 = torch.relu(z1)
        z2 = self.fc2(h1)
        return z2

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
from torchvision import datasets, transforms


class CNN(nn.Module):
    def __init__(self):
        super().__init__()

        self.conv1 = nn.Conv2d(in_channels=1, out_channels=16, kernel_size=3, padding=1)

        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)

        self.fc1 = nn.Linear(16 * 14 * 14, 64)
        self.fc2 = nn.Linear(64, 10)

    def forward(self, x):
        Z = self.conv1(x)
        A = torch.relu(Z)
        P = self.pool(A)
        p = torch.flatten(P, 1)
        z1 = self.fc1(p)
        h1 = torch.relu(z1)
        z2 = self.fc2(h1)
        return z2


transform = transforms.ToTensor()
train_data = datasets.FashionMNIST(
    root="./data", train=True, download=True, transform=transform
)
train_loader = DataLoader(train_data, batch_size=64, shuffle=True)

model = CNN()

criterion = nn.CrossEntropyLoss()

optimizer = optim.Adam(model.parameters(), lr=1e-3)


for epoch in range(10):
    total_loss = 0

    for x, target in train_loader:
        logits = model(x)

        loss = criterion(logits, target)

        optimizer.zero_grad()

        loss.backward()

        optimizer.step()

        total_loss += loss.item()

    print(f"epoch: {epoch + 1}, loss: {total_loss / len(train_loader):.4f}")+ 1}, loss: {total_loss / len(train_loader):.4f}")
